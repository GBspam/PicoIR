cmake_minimum_required(VERSION 3.16)
project(picoir_05 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(picoir STATIC src/ir_protocol.c src/remote_db.c src/signal_analysis.c src/gameboy_ir.c src/gameboy_link.c src/ir_timing.c)
target_include_directories(picoir PUBLIC include)
add_executable(test_ir_protocol tests/test_ir_protocol.c)
target_link_libraries(test_ir_protocol PRIVATE picoir)
add_executable(test_remote_db tests/test_remote_db.c)
target_link_libraries(test_remote_db PRIVATE picoir)
add_executable(test_gameboy_ir tests/test_gameboy_ir.c)
target_link_libraries(test_gameboy_ir PRIVATE picoir)
add_executable(test_console tests/test_console.c src/console.c)
target_link_libraries(test_console PRIVATE picoir)
enable_testing()
add_test(NAME protocol_unit_tests COMMAND test_ir_protocol)
add_test(NAME remote_db_analyzer_tests COMMAND test_remote_db)
add_test(NAME console_tests COMMAND test_console)
add_test(NAME gameboy_ir_packet_tests COMMAND test_gameboy_ir)
add_executable(test_gameboy_link tests/test_gameboy_link.c)
target_link_libraries(test_gameboy_link PRIVATE picoir)
add_test(NAME gameboy_link_state_tests COMMAND test_gameboy_link)
add_executable(test_ir_timing tests/test_ir_timing.c)
target_link_libraries(test_ir_timing PRIVATE picoir)
add_test(NAME ir_timing_classifier_tests COMMAND test_ir_timing)
add_executable(test_signal_samples tests/test_signal_samples.c)
target_link_libraries(test_signal_samples PRIVATE picoir)
add_test(NAME sampled_ir_edge_conversion_tests COMMAND test_signal_samples)
